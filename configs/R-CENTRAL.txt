enable
configure terminal
hostname R-CENTRAL
no ip domain-lookup
banner motd #R-CENTRAL | Frontera OSPF area 0 / EIGRP AS 100#
!
interface Serial0/0/0
 description Enlace a R-OSPF-1 Se0/0/1 | 10.0.0.4/30
 ip address 10.0.0.6 255.255.255.252
 no shutdown
!
interface Serial0/0/1
 description Enlace a R-OSPF-2 Se0/0/1 | 10.0.0.8/30
 ip address 10.0.0.10 255.255.255.252
 no shutdown
!
interface Serial0/1/0
 description Enlace a R-EIGRP-1 Se0/0/1 | 10.0.0.16/30
 ip address 10.0.0.18 255.255.255.252
 no shutdown
!
interface Serial0/1/1
 description Enlace a R-EIGRP-2 Se0/0/1 | 10.0.0.20/30
 ip address 10.0.0.22 255.255.255.252
 no shutdown
!
! --- OSPF area 0 (solo enlaces hacia R-OSPF-1 y R-OSPF-2) ---
router ospf 1
 router-id 3.3.3.3
 network 10.0.0.4 0.0.0.3 area 0
 network 10.0.0.8 0.0.0.3 area 0
!
! --- EIGRP AS 100 (solo enlaces hacia R-EIGRP-1 y R-EIGRP-2) ---
router eigrp 100
 eigrp router-id 3.3.3.3
 no auto-summary
 network 10.0.0.16 0.0.0.3
 network 10.0.0.20 0.0.0.3
!
! --- Redistribucion mutua ---
router eigrp 100
 redistribute ospf 1 metric 1544 2000 255 1 1500
!
router ospf 1
 redistribute eigrp 100 subnets metric-type 1
!
! --- Sumarizacion ---
router ospf 1
 summary-address 172.16.0.0 255.255.255.0
!
interface Serial0/1/0
 ip summary-address eigrp 100 192.168.0.0 255.255.254.0
!
interface Serial0/1/1
 ip summary-address eigrp 100 192.168.0.0 255.255.254.0
!
end
write memory
